cmake_minimum_required(VERSION 3.20)
project(TwinSnakesRecomp LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The packaged launcher currently targets Windows. Use the Python entry point on other platforms.")
endif()
add_executable(TwinSnakes WIN32 src/launcher.cpp)
target_compile_features(TwinSnakes PRIVATE cxx_std_17)
target_compile_definitions(TwinSnakes PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(TwinSnakes PRIVATE shell32)
if(MSVC)
  set_property(TARGET TwinSnakes PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
